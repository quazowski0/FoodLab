const SCALE = 6; // 1 piksel pĹ‚Ăłtna = 6 pikseli ekranu

const reducedMotion = () => window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;

function readPalette() {
  const s = getComputedStyle(document.documentElement);
  const v = (name, fallback) => s.getPropertyValue(name).trim() || fallback;
  return {
    dark: document.documentElement.dataset.theme === 'dark',
    bg: v('--bg', '#f4f8f1'),
    cursor: v('--accent', '#1f9d55'),
    blobs: [v('--blob-1', '#a3e635'), v('--blob-2', '#34d399'), v('--blob-3', '#fde047'), v('--accent', '#1f9d55'), '#14b8a6'],
  };
}

function rgba(hex, a) {
  let h = hex.replace('#', '');
  if (h.length === 3) h = h.split('').map((c) => c + c).join('');
  const n = parseInt(h, 16);
  return `rgba(${(n >> 16) & 255},${(n >> 8) & 255},${n & 255},${a})`;
}

export default function InteractiveBackground() {
  const canvasRef = useRef(null);

  useEffect(() => {
    const root = document.documentElement;
    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');
    const calm = reducedMotion();

    let h = 0;
    let palette = readPalette();
    let frame = 0;
    let running = true;
    let scroll = window.scrollY;

    // kaĹĽda plama ma wĹ‚asny tor (krzywa Lissajous) i tempo
    const blobs = palette.blobs.map((_, i) => ({

      speed: 0.00009 + Math.random() * 0.00008,
      ax: 0.28 + Math.random() * 0.2,
      ay: 0.22 + Math.random() * 0.2,
      size: 0.34 + Math.random() * 0.14,
      depth: 0.15 + i * 0.08, // siĹ‚a parallaxu przy przewijaniu
    }));

    const resize = () => {

      h = Math.max(1, Math.ceil(window.innerHeight / SCALE));
      canvas.width = w;
      canvas.height = h;
    };

    const draw = (t) => {
      const time = calm ? 0 : t;
      const big = Math.max(w, h);
      ctx.globalCompositeOperation = 'source-over';
      ctx.fillStyle = palette.bg;
      ctx.fillRect(0, 0, w, h);
      ctx.globalCompositeOperation = palette.dark ? 'lighter' : 'source-over';

      const alpha = palette.dark ? 0.24 : 0.32;
      blobs.forEach((b, i) => {
        const shift = (scroll * b.depth) / SCALE;
        const x = w * (0.5 + Math.cos(time * b.speed + b.phase) * b.ax);
        let y = h * (0.5 + Math.sin(time * b.speed * 1.3 + b.phase) * b.ay) - shift;
        y = ((y % (h * 1.6)) + h * 1.6) % (h * 1.6) - h * 0.3; // zawijanie przy dĹ‚ugim przewijaniu
        const r = big * b.size;
        const g = ctx.createRadialGradient(x, y, 0, x, y, r);
        g.addColorStop(0, rgba(palette.blobs[i], alpha));
        g.addColorStop(1, rgba(palette.blobs[i], 0));
        ctx.fillStyle = g;
        ctx.fillRect(0, 0, w, h);
      });

      // plama za kursorem
      follow.x += (follow.tx - follow.x) * 0.05;
      follow.y += (follow.ty - follow.y) * 0.05;
      follow.on += ((follow.active ? 1 : 0) - follow.on) * 0.05;
      if (follow.on > 0.01) {
        const x = follow.x * w;
        const y = follow.y * h;
        const r = big * 0.26;
        const g = ctx.createRadialGradient(x, y, 0, x, y, r);
        g.addColorStop(0, rgba(palette.cursor, (palette.dark ? 0.42 : 0.38) * follow.on));
        g.addColorStop(1, rgba(palette.cursor, 0));
        ctx.fillStyle = g;
        ctx.fillRect(0, 0, w, h);

      ctx.globalCompositeOperation = 'source-over';

      if (!running) return;
      draw(t);
      frame = requestAnimationFrame(loop);
    };

    const onMove = (e) => {
      follow.tx = e.clientX / window.innerWidth;
      follow.ty = e.clientY / window.innerHeight;
      if (!follow.active) {
        follow.x = follow.tx;
        follow.y = follow.ty;
        follow.active = true;
      }

      root.style.setProperty('--mouse-y', (follow.ty * 2 - 1).toFixed(3));
    };
    const onLeave = () => {
      follow.active = false;
      root.style.setProperty('--mouse-x', '0');
      root.style.setProperty('--mouse-y', '0');
    };
    const onScroll = () => {
      scroll = window.scrollY;
      root.style.setProperty('--scroll', scroll.toFixed(0));
    };
    const onVisibility = () => {
      cancelAnimationFrame(frame);
      running = !document.hidden;
      if (running) frame = requestAnimationFrame(loop);
    };

    // po zmianie trybu jasny/ciemny pobieramy nowe kolory
    const themeObserver = new MutationObserver(() => {
      palette = readPalette();
    });
    themeObserver.observe(root, { attributes: true, attributeFilter: ['data-theme'] });

    resize();
    onScroll();
    window.addEventListener('resize', resize);
    window.addEventListener('scroll', onScroll, { passive: true });
    window.addEventListener('pointermove', onMove, { passive: true });
    root.addEventListener('pointerleave', onLeave);
    document.addEventListener('visibilitychange', onVisibility);
    frame = requestAnimationFrame(loop);

      running = false;

      themeObserver.disconnect();
      window.removeEventListener('resize', resize);
      window.removeEventListener('scroll', onScroll);
      window.removeEventListener('pointermove', onMove);
      root.removeEventListener('pointerleave', onLeave);

    };

    </div>

  { value: 'keep', label: 'UtrzymaÄ‡ wagÄ™', delta: 0, proteinPerKg: 1.6 },
  { value: 'gain', label: 'PrzytyÄ‡', delta: 300, proteinPerKg: 1.8 },
];

export const DEFAULT_PROFILE = {
  sex: 'm',
  age: 20,
  weight: 70,
  height: 175,
  activity: 1.375,
  goal: 'keep',
  meals: 5,
};

const FAT_SHARE = 0.25; // 25% kalorii z tĹ‚uszczu, reszta po biaĹ‚ku to wÄ™glowodany

function num(value, min, max, fallback) {
  const n = parseFloat(value);
  return Number.isFinite(n) ? Math.min(Math.max(n, min), max) : fallback;
}

export function calcGoals(profile) {
  const age = num(profile.age, 14, 90, DEFAULT_PROFILE.age);
  const weight = num(profile.weight, 35, 200, DEFAULT_PROFILE.weight);
  const height = num(profile.height, 130, 220, DEFAULT_PROFILE.height);
  const activity = num(profile.activity, 1.2, 1.725, DEFAULT_PROFILE.activity);
  const goal = GOAL_TYPES.find((g) => g.value === profile.goal) ?? GOAL_TYPES[1];

  const bmr = 10 * weight + 6.25 * height - 5 * age + (profile.sex === 'k' ? -161 : 5);
  const tdee = bmr * activity;
  const kcal = Math.round((tdee + goal.delta) / 10) * 10;
  const protein = Math.round(goal.proteinPerKg * weight);
  const fat = Math.round((kcal * FAT_SHARE) / 9);
  const carbs = Math.max(Math.round((kcal - protein * 4 - fat * 9) / 4), 0);

  return { kcal, protein, fat, carbs, bmr, tdee, delta: goal.delta, proteinPerKg: goal.proteinPerKg, age, weight, height, activity };
}

export function buildPlan(goalKcal, mealCount) {
  const plan = MEAL_PLANS[mealCount] ?? MEAL_PLANS[5];
  return plan.map(([key, pct]) => ({ key, pct, name: SLOTS[key].name, target: Math.round((goalKcal * pct) / 100) }));

// PosiĹ‚ek z pory, ktĂłrej nie ma w obecnym planie (np. po zmianie 5 na 3), trafia do najbliĹĽszej godzinowo
export function slotOf(meal, plan) {
  if (plan.some((s) => s.key === meal.slot)) return meal.slot;
  const hour = SLOTS[meal.slot]?.hour ?? (parseInt(meal.at, 10) || 12);
  return [...plan].sort((a, b) => Math.abs(SLOTS[a.key].hour - hour) - Math.abs(SLOTS[b.key].hour - hour))[0].key;
}

export function emptySlots(meals, plan) {
  const used = new Set(meals.map((m) => slotOf(m, plan)));
  return plan.filter((s) => !used.has(s.key));
}

// Limit na najbliĹĽszy posiĹ‚ek: pozostaĹ‚e kalorie dzielone proporcjonalnie na pory, ktĂłre jeszcze zostaĹ‚y
export function nextBudget(meals, plan, remainingKcal) {
  const rem = Math.max(remainingKcal, 0);
  const empty = emptySlots(meals, plan);
  if (!empty.length) return { slot: null, kcal: rem };
  const sumPct = empty.reduce((acc, s) => acc + s.pct, 0);
  return { slot: empty[0], kcal: Math.round((rem * empty[0].pct) / sumPct) };
}

// Limit dla konkretnej pory: jeĹ›li jest jeszcze pusta, dostaje proporcjonalnÄ… czÄ™Ĺ›Ä‡ pozostaĹ‚ych kalorii,
// jeĹ›li coĹ› juĹĽ w niej zjedzono, zostaje rĂłĹĽnica do jej celu (ale nie wiÄ™cej niĹĽ reszta dnia).
export function slotBudget(slotKey, meals, plan, remainingKcal) {
  const rem = Math.max(remainingKcal, 0);
  const slot = plan.find((s) => s.key === slotKey);
  if (!slot) return { slot: null, kcal: rem };
  const empty = emptySlots(meals, plan);
  if (empty.some((s) => s.key === slotKey)) {
    const sumPct = empty.reduce((acc, s) => acc + s.pct, 0);
    return { slot, kcal: Math.round((rem * slot.pct) / sumPct) };
  }
  const eaten = meals.filter((m) => slotOf(m, plan) === slotKey).reduce((acc, m) => acc + m.kcal, 0);
  return { slot, kcal: Math.min(Math.max(slot.target - eaten, 0), rem) };
}

// Pora, do ktĂłrej pasuje kategoria: najpierw pusta pora z tej kategorii, potem dowolna z tej kategorii
export function slotForCategory(categoryId, slotToCategory, meals, plan, fallbackKey) {
  const matching = plan.filter((s) => slotToCategory[s.key] === categoryId);
  if (!matching.length) return fallbackKey;
  const empty = new Set(emptySlots(meals, plan).map((s) => s.key));
  return (matching.find((s) => empty.has(s.key)) ?? matching[0]).key;
}

import { useCallback, useState } from 'react';

// Stan zapisywany w localStorage. Odczyt i zapis w try/catch (tryb prywatny, zablokowane dane).
export function useLocalStorage(key, initialValue) {
  const [value, setValue] = useState(() => {
    try {
      const raw = localStorage.getItem(key);
      return raw === null ? initialValue : JSON.parse(raw);
    } catch {
      return initialValue;
    }
  });

  const update = useCallback(
    (next) => {
      setValue((prev) => {
        const resolved = typeof next === 'function' ? next(prev) : next;
        try {
          localStorage.setItem(key, JSON.stringify(resolved));
        } catch {
          /* brak miejsca albo zablokowany localStorage: stan zostaje w pamiÄ™ci */
        }
        return resolved;
      });
    },
    [key],
  );

  return [value, update];
}

// Klucz dnia w czasie lokalnym, np. "2026-11-14"
export function dateKey(date = new Date()) {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

const OFF_URL = 'https://world.openfoodfacts.org/api/v2/product/';
const OFF_FIELDS = 'product_name,product_name_pl,product_name_en,generic_name_pl,brands,quantity,nutriments';

// Odczytuje kod (EAN-13, EAN-8, UPC, Code 128) ze zdjÄ™cia. elementId = pusty, ukryty <div> na stronie.
// Biblioteka Ĺ‚aduje siÄ™ dopiero przy pierwszym skanie, wiÄ™c nie spowalnia startu aplikacji.
export async function readBarcodeFromImage(file, elementId) {
  const { Html5Qrcode, Html5QrcodeSupportedFormats: F } = await import('html5-qrcode');
  const reader = new Html5Qrcode(elementId, {
    verbose: false,
    formatsToSupport: [F.EAN_13, F.EAN_8, F.UPC_A, F.UPC_E, F.CODE_128],
    useBarCodeDetectorIfSupported: true,
  });
  try {
    return await reader.scanFile(file, false);
  } finally {
    reader.clear();
  }
}

export function isValidBarcode(code) {
  return /^\d{8,14}$/.test(String(code).trim());
}

// "500 g" -> { quantity: 500, unit: 'g' }, "1,5 l" -> { quantity: 1.5, unit: 'l' }
export function parsePackage(text) {
  const m = String(text ?? '').toLowerCase().replace(',', '.').match(/(\d+(?:\.\d+)?)\s*(kg|g|ml|cl|l)\b/);
  if (!m) return null;
  const n = Number(m[1]);
  if (m[2] === 'cl') return { quantity: n * 10, unit: 'ml' };
  return { quantity: n, unit: m[2] };
}

function round(n) {
  return Number.isFinite(n) ? Math.round(n * 10) / 10 : undefined;
}

// Zwraca produkt w formacie spiĹĽarni albo null, gdy kodu nie ma w bazie.
// Rzuca bĹ‚Ä…d, gdy nie ma internetu albo API nie odpowiada.
export async function fetchProductByBarcode(code, { signal } = {}) {
  const res = await fetch(`${OFF_URL}${encodeURIComponent(code)}.json?fields=${OFF_FIELDS}`, { signal });
  if (res.status === 404) return null;
  if (!res.ok) throw new Error(`Open Food Facts: HTTP ${res.status}`);
  const data = await res.json();
  if (data.status !== 1 || !data.product) return null;

  const p = data.product;
  const name = p.product_name_pl || p.product_name || p.generic_name_pl || p.product_name_en || '';
  const brand = String(p.brands ?? '').split(',')[0].trim();
  const n = p.nutriments ?? {};
  const kcal = round(n['energy-kcal_100g']);
  const pack = parsePackage(p.quantity);

  return {
    name: brand && name && !name.toLowerCase().includes(brand.toLowerCase()) ? `${name} ${brand}` : name || brand,
    nameEn: p.product_name_en || undefined,
    barcode: String(code),
    source: 'openfoodfacts',
    ...(pack ?? {}),
    nutrition:
      kcal === undefined
        ? undefined
        : {
            kcal,
            protein: round(n.proteins_100g) ?? 0,
            fat: round(n.fat_100g) ?? 0,
            carbs: round(n.carbohydrates_100g) ?? 0,
          },
  };
}

export function makeItem(input) {

export function pantryReducer(items, action) {
  switch (action.type) {
    case 'add': {
      const item = makeItem(action.item);
      if (!item.name) return items;
      // ten sam produkt, ta sama jednostka i data: zwiÄ™kszamy iloĹ›Ä‡ zamiast dublowaÄ‡ wiersz
      const same = items.find(
        (p) => normalizeName(p.name) === normalizeName(item.name) && p.unit === item.unit && p.expiry === item.expiry,
      );
      if (same) {
        return items.map((p) => (p.id === same.id ? { ...p, quantity: toNumber(p.quantity + item.quantity) } : p));
      }
      return [...items, item];
    }
    case 'update':
      return items.map((p) => (p.id === action.id ? makeItem({ ...p, ...action.changes, id: p.id }) : p));
    case 'changeQuantity':
      return items.map((p) =>
        p.id === action.id ? { ...p, quantity: Math.max(0, toNumber(p.quantity + action.delta)) } : p,
      );
    case 'remove':
      return items.filter((p) => p.id !== action.id);
    case 'restore': {
      // cofniÄ™cie usuniÄ™cia: wraca na to samo miejsce
      if (items.some((p) => p.id === action.item.id)) return items;
      const next = [...items];
      next.splice(Math.min(action.index ?? next.length, next.length), 0, action.item);
      return next;
    }
    case 'removeExpired':
      return items.filter((p) => expiryStatus(p.expiry) !== 'expired');
    case 'replace':
      return Array.isArray(action.items) ? action.items.map(makeItem) : items;
    default:
      return items;
  }
}

export function todayKey(date = new Date()) {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');

// Ile dni zostaĹ‚o do koĹ„ca daty (0 = dziĹ›, ujemne = po terminie, null = bez daty)
export function daysUntil(expiry, now = new Date()) {
  if (!expiry) return null;
  const [y, m, d] = expiry.split('-').map(Number);
  const end = new Date(y, m - 1, d);
  const start = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  return Math.round((end - start) / 86400000);
}

export function expiryStatus(expiry, now = new Date()) {
  const days = daysUntil(expiry, now);
  if (days === null) return 'none';
  if (days < 0) return 'expired';
  if (days <= SOON_DAYS) return 'soon';
  return 'ok';
}

function dni(n) {
  return n === 1 ? '1 dzieĹ„' : `${n} dni`;
}

export function expiryLabel(expiry, now = new Date()) {
  const days = daysUntil(expiry, now);
  if (days === null) return 'bez daty waĹĽnoĹ›ci';
  if (days < 0) return `po terminie od ${dni(-days)}`;
  if (days === 0) return 'termin mija dziĹ›';
  if (days === 1) return 'termin mija jutro';
  return `waĹĽne jeszcze ${dni(days)}`;
}

}

import { fetchProductByBarcode, isValidBarcode, readBarcodeFromImage } from '../barcode.js';

const READER_ID = 'sp-barcode-reader';

// Dodawanie przez zdjÄ™cie kodu kreskowego. Na telefonie przycisk od razu otwiera aparat,
// na komputerze moĹĽna wybraÄ‡ zdjÄ™cie z dysku albo wpisaÄ‡ numer spod kodu.
// onFound(produkt) dostaje dane gotowe do wstawienia w formularz.
export default function BarcodeScanner({ onFound }) {
  const fileRef = useRef(null);
  const [manual, setManual] = useState('');
  const [state, setState] = useState({ status: 'idle', message: '' });
  const busy = state.status === 'busy';

  const lookup = async (code) => {
    setState({ status: 'busy', message: `Szukam produktu ${code}...` });
    try {
      const product = await fetchProductByBarcode(code);
      if (product?.name) {
        onFound(product);
        setState({ status: 'ok', message: `Znaleziono: ${product.name}. SprawdĹş iloĹ›Ä‡, ustaw datÄ™ i dodaj.` });
      } else {
        onFound({ barcode: code });
        setState({ status: 'warn', message: `Kodu ${code} nie ma w Open Food Facts. Wpisz nazwÄ™ rÄ™cznie.` });
      }
    } catch {
      onFound({ barcode: code });
      setState({ status: 'error', message: 'Nie udaĹ‚o siÄ™ poĹ‚Ä…czyÄ‡ z Open Food Facts. SprawdĹş internet albo wpisz nazwÄ™ rÄ™cznie.' });
    }
  };

  const onFile = async (e) => {
    const file = e.target.files?.[0];
    e.target.value = ''; // ĹĽeby to samo zdjÄ™cie daĹ‚o siÄ™ wybraÄ‡ jeszcze raz
    if (!file) return;
    setState({ status: 'busy', message: 'OdczytujÄ™ kod ze zdjÄ™cia...' });
    let code;
    try {
      code = await readBarcodeFromImage(file, READER_ID);
    } catch {
      setState({ status: 'error', message: 'Nie widaÄ‡ kodu na zdjÄ™ciu. ZrĂłb ostrzejsze zdjÄ™cie z bliska albo wpisz numer.' });
      return;
    }
    await lookup(code);
  };

  const onManual = (e) => {
    e.preventDefault();
    const code = manual.trim();
    if (!isValidBarcode(code)) {
      setState({ status: 'error', message: 'Numer kodu to 8 do 14 cyfr spod kresek.' });
      return;
    }
    setManual('');
    lookup(code);
  };

  return (
    <div className="sp-scan">
      <div className="sp-scan__row">
        <button type="button" className="sp-btn sp-btn--ghost sp-scan__photo" onClick={() => fileRef.current?.click()} disabled={busy}>
          <svg viewBox="0 0 20 20" aria-hidden="true">
            <path d="M3 5v10M5.5 5v10M8 5v10M11 5v10M13 5v10M15.5 5v10M17 5v10" fill="none" stroke="currentColor" strokeWidth="1.4" strokeLinecap="round" />
          </svg>
          ZdjÄ™cie kodu kreskowego
        </button>
        <input ref={fileRef} type="file" accept="image/*" capture="environment" hidden onChange={onFile} />

        <form className="sp-scan__manual" onSubmit={onManual}>
          <input
            className="sp-input"
            inputMode="numeric"
            value={manual}
            onChange={(e) => setManual(e.target.value.replace(/\D/g, ''))}
            placeholder="albo wpisz numer kodu"
            aria-label="Numer kodu kreskowego"
            maxLength={14}
          />
          <button type="submit" className="sp-btn sp-btn--ghost" disabled={busy}>Szukaj</button>
        </form>
      </div>

      {state.message && (
        <p className={`sp-scan__msg sp-scan__msg--${state.status}`} role="status">{state.message}</p>
      )}
      {/* html5-qrcode potrzebuje elementu w DOM, nawet przy skanowaniu pliku */}
      <div id={READER_ID} hidden />
    </div>
  );
}

// Dodaje klasÄ™ .is-visible, gdy element wjedzie w ekran (animacje przy przewijaniu)
function useReveal() {
  const ref = useRef(null);
  useEffect(() => {
    const root = ref.current;
    if (!root) return;
    const items = root.querySelectorAll('.reveal');
    if (!('IntersectionObserver' in window) || prefersReducedMotion()) {
      items.forEach((el) => el.classList.add('is-visible'));
      return;
    }
    const io = new IntersectionObserver(
      (entries) => {
        entries.forEach((e) => {
          if (e.isIntersecting) {
            e.target.classList.add('is-visible');
            io.unobserve(e.target);
          }
        });
      },
      { threshold: 0.15, rootMargin: '0px 0px -40px 0px' },
    );
    items.forEach((el) => io.observe(el));
    return () => io.disconnect();
  }, []);
  return ref;
}

// Liczba, ktĂłra "odlicza" od zera, gdy pojawi siÄ™ na ekranie
function CountUp({ to, suffix }) {
  const ref = useRef(null);
  // bez animacji (ograniczony ruch / stara przeglÄ…darka) od razu pokazujemy wynik
  const [animate] = useState(
    () => !prefersReducedMotion() && typeof window !== 'undefined' && 'IntersectionObserver' in window,
  );
  const [value, setValue] = useState(animate ? 0 : to);

  useEffect(() => {
    const el = ref.current;
    if (!el || !animate) return;
    let frame;
    const io = new IntersectionObserver(([entry]) => {
      if (!entry.isIntersecting) return;
      io.disconnect();
      const start = performance.now();
      const duration = 1200;
      const tick = (now) => {
        const t = Math.min(1, (now - start) / duration);
        const eased = 1 - Math.pow(1 - t, 3);
        setValue(Math.round(to * eased));
        if (t < 1) frame = requestAnimationFrame(tick);
      };
      frame = requestAnimationFrame(tick);
    });
    io.observe(el);
    return () => {
      io.disconnect();
      cancelAnimationFrame(frame);
    };
  }, [to, animate]);

  return (
    <span ref={ref}>
      {value}
      {suffix}
    </span>
  );
}

// Kafelek, ktĂłry lekko siÄ™ przechyla i podĹ›wietla w miejscu kursora
function TiltCard({ to, children, delay }) {
  const ref = useRef(null);

  const onMove = (e) => {
    const el = ref.current;
    if (!el || prefersReducedMotion()) return;
    const r = el.getBoundingClientRect();
    const x = (e.clientX - r.left) / r.width;
    const y = (e.clientY - r.top) / r.height;
    el.style.setProperty('--mx', `${x * 100}%`);
    el.style.setProperty('--my', `${y * 100}%`);
    el.style.setProperty('--rx', `${(0.5 - y) * 8}deg`);
    el.style.setProperty('--ry', `${(x - 0.5) * 10}deg`);
  };
  const onLeave = () => {
    const el = ref.current;
    if (!el) return;
    el.style.setProperty('--rx', '0deg');
    el.style.setProperty('--ry', '0deg');
  };

  return (
    <Link
      ref={ref}
      to={to}
      className="card reveal"
      style={{ '--delay': `${delay}ms` }}
      onMouseMove={onMove}
      onMouseLeave={onLeave}
    >
      {children}
    </Link>
  );
}

              defaultSlot={targetSlotKey}
              onCook={cook}
              message={message}
            />
          </div>
        </section>
      )}

      {tab === 'cele' && (
        <section className="mk-panel" role="tabpanel" id="mk-pane-cele" aria-labelledby="mk-tab-cele">
          <GoalCalculator profile={profile} goals={goals} plan={plan} onEdit={() => setEditing(true)} />
          <div className="mk-week-box">
            <h2 className="mk-h3">Ostatnie 7 dni</h2>
            <WeekChart days={lastDays(7)} goalKcal={goals.kcal} />
          </div>
        </section>
      )}
    </div>
  );
}
